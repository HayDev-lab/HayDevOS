   951→export function invoiceAgingBucket(invoice: ErpInvoice): "current" | "1_30" | "31_60" | "60plus" {
   952→  if (invoice.status === "paid" || invoice.status === "cancelled" || invoice.status === "draft") return "current";
   953→  if (!invoice.dueAt) return "current";
   954→  const dueMs = new Date(invoice.dueAt).getTime();
   955→  const diffDays = Math.floor((now - dueMs) / 86400000);
   956→  if (diffDays <= 0) return "current";
   957→  if (diffDays <= 30) return "1_30";
   958→  if (diffDays <= 60) return "31_60";
   959→  return "60plus";
   960→}
