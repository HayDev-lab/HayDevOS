     1→     1→/**
     2→     2→ * HayDevOS — Global Activity Timeline feed (Task 13c).
     3→     3→ *
     4→     4→ * A unified, time-ordered activity feed aggregating events from every module's
     5→     5→ * mock data: leads created/stage-changed/SLA-breach, quotes sent/accepted/
     6→     6→ * expiring, documents uploaded/approved/failed, automations ran/succeeded/
     7→     7→ * failed, invoices issued/paid/overdue, payments received, integrations
     8→     8→ * connected/failed/reauth, audit runs completed, task overdue + Owner AI
     9→     9→ * proposals.
    10→    10→ *
    11→    11→ * Each event carries:
    12→    12→ *   - id: stable unique id
    13→    13→ *   - ts: ISO timestamp (string) — sorted descending in the UI. The event ts
    14→    14→ *     is the underlying entity's real timestamp (e.g. lead.createdAt,
    15→    15→ *     automation.runs.lastRunAt, payment.paidAt). Events whose ts falls
    16→    16→ *     outside the configured 7-day window are filtered out by
    17→    17→ *     `buildActivityFeed({ days: 7 })`, so the timeline always shows what
    18→    18→ *     actually happened in the last week.
    19→    19→ *   - module: registry module id ("leados", "quoteflow", …)
    20→    20→ *   - type: domain event type (e.g. "lead.created", "invoice.paid")
    21→    21→ *   - titleKey: i18n key resolved by the sheet UI ("activity.event.lead.created")
    22→    22→ *   - titleParams: interpolation params for that key
    23→    23→ *   - actor: { name, color } — the user or "System" that performed the event
    24→    24→ *   - severity: info | success | warning | critical
    25→    25→ *   - entityId: stable id for the underlying entity (for CSV / drilldown)
    26→    26→ *   - entityLabel: human label (English fallback) shown in tooltips/CSV
    27→    27→ *   - deepLink: { moduleId, context? } — module to switch to when the user
    28→    28→ *      drills down (context may carry an entity id for the module to resolve)
    29→    29→ *
    30→    30→ * `buildActivityFeed(filter)` returns the filtered + sorted feed.
    31→    31→ * `getActivityStats()` returns counts by module/severity/day plus today +
    32→    32→ *   critical totals + modules-touched count — used by the sheet's stats strip.
    33→    33→ *
    34→    34→ * NO indigo / blue — severity colours map to cyan / lime / amber / rose.
    35→    35→ */
    36→    36→
    37→    37→export type ActivitySeverity = "info" | "success" | "warning" | "critical";
    38→    38→
    39→    39→export interface ActivityActor {
    40→    40→  name: string;
    41→    41→  /** Tailwind accent token used for the avatar chip. */
    42→    42→  color: "lime" | "cyan" | "amber" | "rose" | "violet";
    43→    43→  /** Pre-computed initials ("Aram H" → "AH") for the avatar chip. */
    44→    44→  initials: string;
    45→    45→}
    46→    46→
    47→    47→export interface ActivityDeepLink {
    48→    48→  moduleId: string;
    49→    49→  /** Optional entity context (e.g. "ld_005"). Modules can use this to scroll/highlight. */
    50→    50→  context?: string;
    51→    51→}

    67→    67→// ─────────────────────────────────────────────────────────────────────────────
    68→    68→// Time helpers (kept local to keep this module self-contained). The "now"
    69→    69→// value is captured once at module load so the relative ordering stays stable
    70→    70→// across re-renders within a session — the timeline always shows events
    71→    71→// "spread across the last 7 days" relative to that anchor.
    72→    72→// ─────────────────────────────────────────────────────────────────────────────
    73→    73→
    74→    74→const NOW = Date.now();
    75→    75→const MS_DAY = 86_400_000;
    76→    76→const MS_HOUR = 3_600_000;
    77→    77→const MS_MIN = 60_000;
    78→    78→const WINDOW_DAYS = 7;
    79→    79→const WINDOW_MS = WINDOW_DAYS * MS_DAY;
    80→    80→
    81→    81→const iso = (msAgo: number) => new Date(NOW - msAgo).toISOString();
    82→    82→const daysAgoMs = (d: number) => d * MS_DAY;
    83→    83→const hoursAgoMs = (h: number) => h * MS_HOUR;
    84→    84→const minsAgoMs = (m: number) => m * MS_MIN;

    90→    90→function makeActor(name: string, color: ActivityActor["color"]): ActivityActor {
    91→    91→  const parts = name.split(" ").filter(Boolean).slice(0, 2);
    92→    92→  const init = parts.map((p) => p[0]?.toUpperCase() ?? "").join("") || "?";
    93→    93→  return { name, color, initials: init };
    94→    94→}

   106→   106→interface CurrencyFmt {
   107→   107→  amount: number;
   108→   108→  currency: string;
   109→   109→}
   110→   110→
   111→   111→function money(v: CurrencyFmt): string {
   112→   112→  const symbols: Record<string, string> = { USD: "$", EUR: "€", RUB: "₽", AMD: "֏", GBP: "£", JPY: "¥" };
   113→   113→  const sym = symbols[v.currency] ?? "";
   114→   114→  const num = new Intl.NumberFormat("en-US", {
   115→   115→    minimumFractionDigits: v.amount % 1 === 0 ? 0 : 2,
   116→   116→    maximumFractionDigits: 2,
   117→   117→  }).format(v.amount);
   118→   118→  return `${sym}${num}`;
   119→   119→}

   144→   144→function buildLeadEvents(): ActivityEvent[] {
   145→   145→  const out: ActivityEvent[] = [];
   146→   146→  for (const l of mockLeads) {
   147→   147→    // created — uses the lead's real createdAt.
   148→   148→    out.push({
   149→   149→      id: `act_lead_created_${l.id}`,
   150→   150→      ts: l.createdAt,
   151→   151→      module: "leados",
   152→   152→      type: "lead.created",
   153→   153→      titleKey: "activity.event.lead.created",
   154→   154→      titleParams: { name: l.name },
   155→   155→      actor: actorForOwnerId(l.ownerId),
   156→   156→      severity: "info",
   157→   157→      entityId: l.id,
   158→   158→      entityLabel: l.name,
   159→   159→      deepLink: { moduleId: "leados", context: l.id },
   160→   160→    });
   161→   161→
   162→   162→    // stage changed — fired on the lead's lastActivityAt.
   163→   163→    if (l.stage !== "new") {
   164→   164→      out.push({
   165→   165→        id: `act_lead_stage_${l.id}`,
   166→   166→        ts: l.lastActivityAt,
   167→   167→        module: "leados",
   168→   168→        type: "lead.stage_changed",
   169→   169→        titleKey: "activity.event.lead.stage_changed",
   170→   170→        titleParams: { name: l.name, stage: stageLabel(l.stage) },
   171→   171→        actor: actorForOwnerId(l.ownerId),
   172→   172→        severity: l.stage === "won" ? "success" : l.stage === "lost" ? "warning" : "info",
   173→   173→        entityId: l.id,
   174→   174→        entityLabel: `${l.name} · ${stageLabel(l.stage)}`,
   175→   175→        deepLink: { moduleId: "leados", context: l.id },
   176→   176→      });
   177→   177→    }
   178→   178→
   179→   179→    // sla breach
   180→   180→    if (l.slaBreached) {
   181→   181→      out.push({
   182→   182→        id: `act_lead_sla_${l.id}`,
   183→   183→        ts: l.slaDueAt ?? l.lastActivityAt,
   184→   184→        module: "leados",
   185→   185→        type: "lead.sla_breach",
   186→   186→        titleKey: "activity.event.lead.sla_breach",
   187→   187→        titleParams: { name: l.name },
   188→   188→        actor: ACTOR_SYSTEM,
   189→   189→        severity: "critical",
   190→   190→        entityId: l.id,
   191→   191→        entityLabel: l.name,
   192→   192→        deepLink: { moduleId: "leados", context: l.id },
   193→   193→      });
   194→   194→    }
   195→   195→  }
   196→   196→  return out;
   197→   197→}

   199→   199→function buildQuoteEvents(): ActivityEvent[] {
   200→   200→  const out: ActivityEvent[] = [];
   201→   201→  for (const q of mockQuotes) {
   202→   202→    const total = money({ amount: q.total, currency: q.currency });
   203→   203→    // The "sent/accepted/expiring" event ts is the quote's real createdAt
   204→   204→    // (sent) or a synthetic recent ts (accepted — no separate field on the
   205→   205→    // mock quote).
   206→   206→    if (q.status === "sent") {
   207→   207→      // expiring soon? if validUntil is within 3 days from now, flag it.
   208→   208→      const validMs = new Date(q.validUntil).getTime();
   209→   209→      const daysToExpiry = (validMs - NOW) / MS_DAY;
   210→   210→      if (daysToExpiry >= 0 && daysToExpiry <= 3) {
   211→   211→        out.push({
   212→   212→          id: `act_quote_expiring_${q.id}`,
   213→   213→          ts: iso(hoursAgoMs(2)),
   214→   214→          module: "quoteflow",
   215→   215→          type: "quote.expiring",
   216→   216→          titleKey: "activity.event.quote.expiring",
   217→   217→          titleParams: { number: q.number, total },
   218→   218→          actor: ACTOR_REP1,
   219→   219→          severity: "warning",
   220→   220→          entityId: q.id,
   221→   221→          entityLabel: q.number,
   222→   222→          deepLink: { moduleId: "quoteflow", context: q.id },
   223→   223→        });
   224→   224→      }
   225→   225→      out.push({
   226→   226→        id: `act_quote_sent_${q.id}`,
   227→   227→        ts: q.createdAt,
   228→   228→        module: "quoteflow",
   229→   229→        type: "quote.sent",
   230→   230→        titleKey: "activity.event.quote.sent",
   231→   231→        titleParams: { number: q.number, total },
   232→   232→        actor: ACTOR_REP1,
   233→   233→        severity: "info",
   234→   234→        entityId: q.id,
   235→   235→        entityLabel: q.number,
   236→   236→        deepLink: { moduleId: "quoteflow", context: q.id },
   237→   237→      });
   238→   238→    } else if (q.status === "accepted") {
   239→   239→      // The accepted event ts is taken to be ~1d after the quote was created
   240→   240→      // (synthetic — the mock data only stores `createdAt`).
   241→   241→      const acceptedTs = new Date(
   242→   242→        new Date(q.createdAt).getTime() + MS_DAY,
   243→   243→      ).toISOString();
   244→   244→      out.push({
   245→   245→        id: `act_quote_accepted_${q.id}`,
   246→   246→        ts: acceptedTs,
   247→   247→        module: "quoteflow",
   248→   248→        type: "quote.accepted",
   249→   249→        titleKey: "activity.event.quote.accepted",
   250→   250→        titleParams: { number: q.number, total },
   251→   251→        actor: ACTOR_OWNER,
   252→   252→        severity: "success",
   253→   253→        entityId: q.id,
   254→   254→        entityLabel: q.number,
   255→   255→        deepLink: { moduleId: "quoteflow", context: q.id },
   256→   256→      });
   257→   257→    }
   258→   258→    // draft / rejected / expired — skip; they don't surface as timeline events
   259→   259→    // (or surface as different types in downstream tasks).
   260→   260→  }
   261→   261→  return out;
   262→   262→}

   321→   321→function buildAutomationEvents(): ActivityEvent[] {
   322→   322→  const out: ActivityEvent[] = [];
   323→   323→  for (const a of mockAutomations) {
   324→   324→    // Skip automations that have never run (drafts with 0 executions) — they
   325→   325→    // don't surface as activity yet.
   326→   326→    if (a.runs.total === 0) continue;
   327→   327→    // The most recent run (or created-at as fallback).
   328→   328→    const lastRunTs = a.runs.lastRunAt ?? a.createdAt;
   329→   329→    // Each automation emits a single run event — failed runs surface as
   330→   330→    // critical, succeeded runs as success.
   331→   331→    const hasFailures = a.runs.failed > 0;
   332→   332→    out.push({
   333→   333→      id: `act_auto_run_${a.id}`,
   334→   334→      ts: lastRunTs,
   335→   335→      module: "autopilot",
   336→   336→      type: hasFailures ? "automation.failed" : "automation.succeeded",
   337→   337→      titleKey: hasFailures
   338→   338→        ? "activity.event.automation.failed"
   339→   339→        : "activity.event.automation.succeeded",
   340→   340→      titleParams: { name: a.name },
   341→   341→      actor: ACTOR_SYSTEM,
   342→   342→      severity: hasFailures ? "critical" : "success",
   343→   343→      entityId: a.id,
   344→   344→      entityLabel: a.name,
   345→   345→      deepLink: { moduleId: "autopilot", context: a.id },
   346→   346→    });
   347→   347→  }
   348→   348→  return out;
   349→   349→}

   625→   625→// ─────────────────────────────────────────────────────────────────────────────
   626→   626→// Filter API
   627→   627→// ─────────────────────────────────────────────────────────────────────────────
   628→   628→
   629→   629→export interface ActivityFilter {
   630→   630→  /** Module id ("leados", "quoteflow", …) — undefined means "all modules". */
   631→   631→  module?: string;
   632→   632→  /** Severity filter — undefined means "all severities". */
   633→   633→  severity?: ActivitySeverity;
   634→   634→  /** Time window in days — default 7. */
   635→   635→  days?: number;
   636→   636→  /** Free-text search across title params, actor name, entity label. */
   637→   637→  query?: string;
   638→   638→}
   639→   639→
   640→   640→/** Match a single event against the filter, applied to the *English* label
   641→   641→ *  text. The timeline UI re-resolves the titleKey against the live locale so
   642→   642→ *  the actual rendered text is localized; the search matches on the stable
   643→   643→ *  English fallback fields so results are deterministic across locales. */
   644→   644→function matchesFilter(ev: ActivityEvent, filter: ActivityFilter): boolean {
   645→   645→  if (filter.module && ev.module !== filter.module) return false;
   646→   646→  if (filter.severity && ev.severity !== filter.severity) return false;
   647→   647→  const days = filter.days ?? WINDOW_DAYS;
   648→   648→  const cutoff = NOW - days * MS_DAY;
   649→   649→  const tsMs = new Date(ev.ts).getTime();
   650→   650→  if (tsMs < cutoff) return false;
   651→   651→  const q = filter.query?.trim().toLowerCase();
   652→   652→  if (q) {
   653→   653→    const hay = [
   654→   654→      ev.entityLabel,
   655→   655→      ev.actor.name,
   656→   656→      ev.module,
   657→   657→      ev.type,
   658→   658→      ev.entityId,
   659→   659→    ]
   660→   660→      .join(" ")
   661→   661→      .toLowerCase();
   662→   662→    if (!hay.includes(q)) return false;
   663→   663→  }
   664→   664→  return true;
   665→   665→}
   666→   666→
   667→   667→/** Build (filter) the activity feed. Always returns a fresh array (sorted
   668→   668→ *  most-recent-first) — safe to memoize on the caller side keyed by the
   669→   669→ *  filter object's stable fields. */
   670→   670→export function buildActivityFeed(filter: ActivityFilter = {}): ActivityEvent[] {
   671→   671→  const all = getActivityFeed();
   672→   672→  return all.filter((ev) => matchesFilter(ev, filter));
   673→   673→}

   675→   675→// ─────────────────────────────────────────────────────────────────────────────
   676→   676→// Stats — counts by module, severity, and day, plus today/critical/modulesTouched
   677→   677→// totals. Used by the sheet's compact stats strip above the timeline.
   678→   678→// ─────────────────────────────────────────────────────────────────────────────
   679→   679→
   680→   680→export interface ActivityStats {
   681→   681→  total: number;
   682→   682→  critical: number;
   683→   683→  today: number;
   684→   684→  modulesTouched: number;
   685→   685→  byModule: Record<string, number>;
   686→   686→  bySeverity: Record<ActivitySeverity, number>;
   687→   687→  /** Per-day counts keyed by YYYY-MM-DD (local). Index 0 is today. */
   688→   688→  byDay: { date: string; count: number }[];
   689→   689→}
   690→   690→
   691→   691→/** Compute stats from the full feed (filtered to the 7-day window). */
   692→   692→export function getActivityStats(): ActivityStats {
   693→   693→  const all = getActivityFeed();
   694→   694→  const cutoff = NOW - WINDOW_MS;
   695→   695→  const inWindow = all.filter((ev) => {
   696→   696→    const t = new Date(ev.ts).getTime();
   697→   697→    return !isNaN(t) && t >= cutoff;
   698→   698→  });
   699→   699→
   700→   700→  const byModule: Record<string, number> = {};
   701→   701→  const bySeverity: Record<ActivitySeverity, number> = {
   702→   702→    info: 0,
   703→   703→    success: 0,
   704→   704→    warning: 0,
   705→   705→    critical: 0,
   706→   706→  };
   707→   707→  const dayCounts = new Map<string, number>();
   708→   708→
   709→   709→  // Today & yesterday key helpers (local-time).
   710→   710→  const todayDate = new Date();
   711→   711→  const todayKey = todayDate.toDateString();
   712→   712→
   713→   713→  let today = 0;
   714→   714→  let critical = 0;
   715→   715→
   716→   716→  for (const ev of inWindow) {
   717→   717→    byModule[ev.module] = (byModule[ev.module] ?? 0) + 1;
   718→   718→    bySeverity[ev.severity] += 1;
   719→   719→    if (ev.severity === "critical") critical += 1;
   720→   720→    const d = new Date(ev.ts);
   721→   721→    if (!isNaN(d.getTime())) {
   722→   722→      const key = d.toDateString();
   723→   723→      if (key === todayKey) today += 1;
   724→   724→      dayCounts.set(key, (dayCounts.get(key) ?? 0) + 1);
   725→   725→    }
   726→   726→  }
   727→   727→
   728→   728→  // Build per-day series (oldest → newest) for a 7-day sparkline.
   729→   729→  const byDay: { date: string; count: number }[] = [];
   730→   730→  for (let i = WINDOW_DAYS - 1; i >= 0; i--) {
   731→   731→    const d = new Date(NOW - i * MS_DAY);
   732→   732→    byDay.push({ date: d.toDateString(), count: dayCounts.get(d.toDateString()) ?? 0 });
   733→   733→  }
   734→   734→
   735→   735→  return {
   736→   736→    total: inWindow.length,
   737→   737→    critical,
   738→   738→    today,
   739→   739→    modulesTouched: Object.keys(byModule).length,
   740→   740→    byModule,
   741→   741→    bySeverity,
   742→   742→    byDay,
   743→   743→  };
   744→   744→}

   746→   746→// ─────────────────────────────────────────────────────────────────────────────
   747→   747→// Day grouping helpers — used by the sheet UI.
   748→   748→// ─────────────────────────────────────────────────────────────────────────────
   749→   749→
   750→   750→export interface ActivityDayGroup {
   751→   751→  /** toDateString() of the day. */
   752→   752→  key: string;
   753→   753→  /** Pre-localized display label ("Today" | "Yesterday" | null for raw date format). */
   754→   754→  labelKey: "activity.day.today" | "activity.day.yesterday" | null;
   755→   755→  events: ActivityEvent[];
   756→   756→}
   757→   757→
   758→   758→/** Group events into day buckets (most recent first), preserving per-event
   759→   759→ *  ordering inside each bucket. */
   760→   760→export function groupByDay(events: ActivityEvent[]): ActivityDayGroup[] {
   761→   761→  const buckets = new Map<string, ActivityEvent[]>();
   762→   762→  const today = new Date();
   763→   763→  const todayKey = today.toDateString();
   764→   764→  const yesterdayKey = new Date(today.getTime() - MS_DAY).toDateString();
   765→   765→  for (const ev of events) {
   766→   766→    const d = new Date(ev.ts);
   767→   767→    const key = d.toDateString();
   768→   768→    const arr = buckets.get(key) ?? [];
   769→   769→    arr.push(ev);
   770→   770→    buckets.set(key, arr);
   771→   771→  }
   772→   772→  // Preserve the order events were given to us (already sorted desc) when
   773→   773→  // building the bucket list, since Map preserves insertion order.
   774→   774→  const out: ActivityDayGroup[] = [];
   775→   775→  for (const [key, evs] of buckets) {
   776→   776→    let labelKey: ActivityDayGroup["labelKey"] = null;
   777→   777→    if (key === todayKey) labelKey = "activity.day.today";
   778→   778→    else if (key === yesterdayKey) labelKey = "activity.day.yesterday";
   779→   779→    out.push({ key, labelKey, events: evs });
   780→   780→  }
   781→   781→  return out;
   782→   782→}

   785→   785→// CSV export — produces a simple CSV string from the (filtered) events.

   788→   788→function csvEscape(value: string): string {
   789→   789→  if (value.includes(",") || value.includes("\"") || value.includes("\n")) {
   790→   790→    return `"${value.replace(/"/g, '""')}"`;
   791→   791→  }
   792→   792→  return value;
   793→   793→}
