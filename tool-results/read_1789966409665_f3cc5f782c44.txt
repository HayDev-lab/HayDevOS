   178→// ─────────────────────────────────────────────────────────────────────────────
   179→// AR aging
   180→// ─────────────────────────────────────────────────────────────────────────────
   181→
   182→function ArAgingReport() {
   183→  const { t, locale } = useLocale();
   184→  const rows = useMemo(() => {
   185→    return erpInvoices
   186→      .filter((inv) => inv.status !== "paid" && inv.status !== "cancelled" && inv.status !== "draft")
   187→      .map((inv) => {
   188→        const aging = invoiceAgingBucket(inv);
   189→        const outstanding = Math.max(0, inv.amount - inv.paidAmount);
   190→        return {
   191→          inv,
   192→          customer: customerName(inv.customerId),
   193→          aging,
   194→          outstanding,
   195→        };
   196→      })
   197→      .sort((a, b) => {
   198→        const order: Record<string, number> = { "60plus": 0, "31_60": 1, "1_30": 2, current: 3 };
   199→        return (order[a.aging] ?? 4) - (order[b.aging] ?? 4);
   200→      });
   201→  }, []);
   202→
   203→  const buckets = useMemo(() => {
   204→    const b = { current: 0, d1_30: 0, d31_60: 0, d60plus: 0 };
   205→    for (const r of rows) {
   206→      if (r.aging === "current") b.current += r.outstanding;
   207→      else if (r.aging === "1_30") b.d1_30 += r.outstanding;
   208→      else if (r.aging === "31_60") b.d31_60 += r.outstanding;
   209→      else b.d60plus += r.outstanding;
   210→    }
   211→    return b;
   212→  }, [rows]);
   213→
   214→  const chartData = [
   215→    { name: t("erp.aging.current"), value: buckets.current, tone: "success" as const },
   216→    { name: t("erp.aging.1_30"), value: buckets.d1_30, tone: "warning" as const },
   217→    { name: t("erp.aging.31_60"), value: buckets.d31_60, tone: "amber" as const },
   218→    { name: t("erp.aging.60plus"), value: buckets.d60plus, tone: "destructive" as const },
   219→  ];
   220→
   221→  const AGING_COLORS: Record<string, string> = {
   222→    success: "var(--success)",
   223→    warning: "var(--warning)",
   224→    amber: "var(--accent-amber)",
   225→    destructive: "var(--destructive)",
   226→  };
   227→
   228→  return (
   229→    <div className="grid grid-cols-1 gap-4 lg:grid-cols-2">
   230→      <Card className="surface-elevated">
   231→        <CardContent className="p-0">
   232→          <div className="border-b border-border px-5 py-3">
   233→            <h3 className="text-sm font-semibold text-foreground">{t("erp.report.arAging")}</h3>
   234→          </div>
   235→          <ScrollArea className="max-h-[420px]">
   236→            <Table>
   237→              <TableHeader>
   238→                <TableRow>
   239→                  <TableHead>{t("erp.invoice.number")}</TableHead>
   240→                  <TableHead>{t("erp.invoice.customer")}</TableHead>
   241→                  <TableHead className="text-right">{t("erp.invoice.outstanding")}</TableHead>
   242→                  <TableHead>{t("erp.invoice.aging")}</TableHead>
   243→                </TableRow>
   244→              </TableHeader>
   245→              <TableBody>
   246→                {rows.length === 0 ? (
   247→                  <TableRow><TableCell colSpan={4} className="py-8 text-center text-sm text-muted-foreground">—</TableCell></TableRow>
   248→                ) : (
   249→                  rows.map((r) => {
   250→                    const tone =
   251→                      r.aging === "current" ? "success" :
   252→                      r.aging === "1_30" ? "warning" :
   253→                      r.aging === "31_60" ? "warning" :
   254→                      "destructive";
   255→                    const cls = toneClasses(tone);
   256→                    const labelKey =
   257→                      r.aging === "current" ? "erp.aging.current" :
   258→                      r.aging === "1_30" ? "erp.aging.1_30" :
   259→                      r.aging === "31_60" ? "erp.aging.31_60" :
   260→                      "erp.aging.60plus";
   261→                    return (
   262→                      <TableRow key={r.inv.id}>
   263→                        <TableCell className="font-mono text-xs font-medium">{r.inv.number}</TableCell>
   264→                        <TableCell className="text-sm">{r.customer}</TableCell>
   265→                        <TableCell className={cn("text-right text-sm font-medium", cls.text)}>
   266→                          {formatCurrency(r.outstanding, r.inv.currency)}
   267→                        </TableCell>
   268→                        <TableCell>
   269→                          <Badge variant="outline" className={cn("px-1.5 py-0 text-[10px] uppercase tracking-wider", cls.border, cls.bg, cls.text)}>
   270→                            {t(labelKey)}
   271→                          </Badge>
   272→                        </TableCell>
   273→                      </TableRow>
   274→                    );
   275→                  })
   276→                )}
   277→              </TableBody>
   278→            </Table>
   279→          </ScrollArea>
   280→        </CardContent>
   281→      </Card>
   282→
   283→      <Card className="surface-elevated">
   284→        <CardContent className="p-0">
   285→          <div className="border-b border-border px-5 py-3">
   286→            <h3 className="text-sm font-semibold text-foreground">{t("erp.report.arAging")}</h3>
   287→          </div>
   288→          <div className="h-[300px] w-full p-4">
   289→            <ResponsiveContainer width="100%" height="100%">
   290→              <BarChart data={chartData} margin={{ top: 8, right: 8, bottom: 0, left: 8 }}>
   291→                <CartesianGrid stroke="var(--border)" strokeDasharray="3 3" vertical={false} />
   292→                <XAxis dataKey="name" tick={{ fill: "var(--muted-foreground)", fontSize: 11 }} axisLine={{ stroke: "var(--border)" }} tickLine={false} />
   293→                <YAxis tick={{ fill: "var(--muted-foreground)", fontSize: 11 }} axisLine={false} tickLine={false} tickFormatter={(v) => formatCompact(v as number)} />
   294→                <Tooltip
   295→                  contentStyle={{ background: "var(--popover)", border: "1px solid var(--border)", borderRadius: 8, color: "var(--popover-foreground)", fontSize: 12 }}
   296→                  formatter={(value: number) => [formatCurrency(value), t("erp.invoice.outstanding")]}
   297→                  cursor={{ fill: "color-mix(in oklab, var(--muted-foreground) 8%, transparent)" }}
   298→                />
   299→                <Bar dataKey="value" radius={[6, 6, 0, 0]} maxBarSize={64}>
   300→                  {chartData.map((entry, idx) => (
   301→                    <Cell key={idx} fill={AGING_COLORS[entry.tone]} />
   302→                  ))}
   303→                </Bar>
   304→              </BarChart>
   305→            </ResponsiveContainer>
   306→          </div>
   307→        </CardContent>
   308→      </Card>
   309→    </div>
   310→  );
   311→}

   317→function SalesByProductReport() {

   321→    const map = new Map<string, { name: string; qty: number; revenue: number }>();

   324→        const cur = map.get(li.description) ?? { name: li.description, qty: 0, revenue: 0 };

   472→// ─────────────────────────────────────────────────────────────────────────────
   473→// Inventory valuation
   474→// ─────────────────────────────────────────────────────────────────────────────
   475→
   476→function InventoryValuationReport() {
   477→  const { t } = useLocale();
   478→
   479→  const rows = useMemo(() => {
   480→    return erpProducts
   481→      .filter((p) => p.reorderPoint > 0 || p.stock < 1000)
   482→      .map((p) => ({
   483→        product: p,
   484→        qty: p.stock,
   485→        unitCost: p.cost,
   486→        valuation: p.stock * p.cost,
   487→        margin: productMarginPct(p),
   488→      }))
   489→      .sort((a, b) => b.valuation - a.valuation);
   490→  }, []);
   491→
   492→  const totalValuation = rows.reduce((s, r) => s + r.valuation, 0);
   493→
   494→  const chartData = rows.slice(0, 8).map((r) => ({ name: r.product.sku, value: r.valuation }));
   495→
   496→  return (
