   115→function PipelineTab() {
   116→  const { t } = useLocale();
   117→  const settings = useSettings();
   118→  const createStage = useCreateStage();
   119→  const updateStage = useUpdateStage();
   120→  const delStage = useDeleteStage();
   121→  const [newStage, setNewStage] = useState<{ name: string; type: string; color: string }>({ name: "", type: "open", color: "#94a3b8" });
   122→
   123→  const sensors = useSensors(
   124→    useSensor(PointerSensor, { activationConstraint: { distance: 4 } }),
   125→    useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates })
   126→  );
   127→
   128→  if (settings.isLoading) return <Skeleton className="h-48 w-full" />;
   129→  const pipelines = settings.data?.pipelines ?? [];
   130→
   131→  const addStage = async (pipelineId: string) => {
   132→    if (!newStage.name.trim()) return;
   133→    try {
   134→      await createStage.mutateAsync({ pipelineId, name: newStage.name, type: newStage.type, color: newStage.color });
   135→      toast.success("Stage added");
   136→      setNewStage({ name: "", type: "open", color: "#94a3b8" });
   137→    } catch (e) { toast.error((e as Error).message); }
   138→  };
   139→  const renameStage = async (id: string, name: string) => {
   140→    try { await updateStage.mutateAsync({ id, body: { name } }); } catch (e) { toast.error((e as Error).message); }
   141→  };
   142→  const recolorStage = async (id: string, color: string) => {
   143→    try { await updateStage.mutateAsync({ id, body: { color } }); } catch (e) { toast.error((e as Error).message); }
   144→  };
   145→  const removeStage = async (id: string) => {
   146→    try { await delStage.mutateAsync(id); toast.success("Stage deleted"); } catch (e) { toast.error((e as Error).message); }
   147→  };
   148→  const onDragEnd = async (pipelineId: string, event: DragEndEvent) => {
   149→    const { active, over } = event;
   150→    if (!over || active.id === over.id) return;
   151→    const p = pipelines.find((pp: any) => pp.id === pipelineId);
   152→    if (!p) return;
   153→    const oldIndex = p.stages.findIndex((s: any) => s.id === active.id);
   154→    const newIndex = p.stages.findIndex((s: any) => s.id === over.id);
   155→    if (oldIndex === -1 || newIndex === -1) return;
   156→    const reordered = arrayMove(p.stages as any[], oldIndex, newIndex) as any[];
   157→    // persist new positions
   158→    for (let i = 0; i < reordered.length; i++) {
   159→      if (reordered[i].position !== i) {
   160→        try { await updateStage.mutateAsync({ id: reordered[i].id, body: { position: i } }); } catch {}
   161→      }
   162→    }
   163→    toast.success("Stages reordered");
   164→  };
   165→
   166→  return (
   167→    <div className="space-y-3">
   168→      {pipelines.map((p: any) => (
   169→        <Card key={p.id}>
   170→          <CardHeader className="pb-2 flex flex-row items-center justify-between">
   171→            <CardTitle className="text-sm">{p.name}</CardTitle>
   172→            {p.isDefault && <Badge>Default</Badge>}
   173→          </CardHeader>
   174→          <CardContent className="pt-0 space-y-2">
   175→            <DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={(e) => onDragEnd(p.id, e)}>
   176→              <SortableContext items={p.stages.map((s: any) => s.id)} strategy={verticalListSortingStrategy}>
   177→                <div className="space-y-1.5">
   178→                  {p.stages.map((s: any) => (
   179→                    <SortableStage
   180→                      key={s.id}
   181→                      stage={s}
   182→                      onRecolor={recolorStage}
   183→                      onRename={renameStage}
   184→                      onRemove={removeStage}
   185→                      updating={updateStage.isPending}
   186→                    />
   187→                  ))}
   188→                </div>
   189→              </SortableContext>
   190→            </DndContext>
   191→            {/* add new stage */}
   192→            <div className="flex items-center gap-2 pt-1 border-t">
   193→              <input
   194→                type="color"
   195→                value={newStage.color}
   196→                onChange={(e) => setNewStage((s) => ({ ...s, color: e.target.value }))}
   197→                className="h-6 w-6 rounded cursor-pointer border-0 bg-transparent p-0"
   198→              />
   199→              <Input
   200→                value={newStage.name}
   201→                onChange={(e) => setNewStage((s) => ({ ...s, name: e.target.value }))}
   202→                placeholder="New stage name…"
   203→                className="h-8 flex-1 text-sm"
   204→              />
   205→              <Select value={newStage.type} onValueChange={(v) => setNewStage((s) => ({ ...s, type: v }))}>
   206→                <SelectTrigger className="h-8 w-24 text-xs"><SelectValue /></SelectTrigger>
   207→                <SelectContent>
   208→                  <SelectItem value="open">open</SelectItem>
   209→                  <SelectItem value="won">won</SelectItem>
   210→                  <SelectItem value="lost">lost</SelectItem>
   211→                </SelectContent>
   212→              </Select>
   213→              <Button size="sm" variant="outline" onClick={() => addStage(p.id)} disabled={!newStage.name.trim() || createStage.isPending}>
   214→                <Plus className="h-3.5 w-3.5 mr-1" />{t("common.create")}
   215→              </Button>
   216→            </div>
   217→          </CardContent>
   218→        </Card>
   219→      ))}
   220→      <p className="text-xs text-muted-foreground px-1 flex items-center gap-1.5">
   221→        <GripVertical className="h-3 w-3" />
   222→        Drag the handle to reorder · rename inline · recolor via swatch · delete (blocked if leads are in the stage). Changes reflect immediately in the Kanban.
   223→      </p>
   224→    </div>
   225→  );
   226→}

   228→function SortableStage({ stage, onRecolor, onRename, onRemove, updating }: { stage: any; onRecolor: (id: string, c: string) => void; onRename: (id: string, n: string) => void; onRemove: (id: string) => void; updating: boolean }) {
   229→  const { attributes, listeners, setNodeRef, transform, transition, isDragging } = useSortable({ id: stage.id });
   230→  const style = {
   231→    transform: CSS.Transform.toString(transform),
   232→    transition,
   233→    zIndex: isDragging ? 10 : "auto",
   234→  };
   235→  return (
   236→    <div ref={setNodeRef} style={style} className={cn("flex items-center gap-2 rounded-lg border px-2 py-1.5 bg-card", isDragging && "shadow-lg ring-2 ring-primary/30 opacity-90")}>
   237→      <button {...attributes} {...listeners} className="cursor-grab active:cursor-grabbing text-muted-foreground hover:text-foreground touch-none" title="Drag to reorder">
   238→        <GripVertical className="h-4 w-4" />
   239→      </button>
   240→      <input
   241→        type="color"
   242→        value={stage.color ?? "#94a3b8"}
   243→        onChange={(e) => onRecolor(stage.id, e.target.value)}
   244→        className="h-6 w-6 rounded cursor-pointer border-0 bg-transparent p-0"
   245→        title="Stage color"
   246→      />
   247→      <input
   248→        defaultValue={stage.name}
   249→        onBlur={(e) => { if (e.target.value !== stage.name) onRename(stage.id, e.target.value); }}
   250→        className="flex-1 bg-transparent text-sm font-medium outline-none border-b border-transparent focus:border-primary"
   251→      />
   252→      <Badge variant="outline" className="text-[10px] px-1 py-0">{stage.type}</Badge>
   253→      {updating && <Loader2 className="h-3 w-3 animate-spin text-muted-foreground" />}
   254→      <button
   255→        onClick={() => onRemove(stage.id)}
   256→        className="text-muted-foreground hover:text-red-500 text-xs px-1"
   257→        title="Delete stage"
   258→      >✕</button>
   259→    </div>
   260→  );
   261→}

   383→function CustomFieldsTab() {
   384→  const { t } = useLocale();
   385→  const fields = useCustomFields();
   386→  const create = useCreateCustomField();
   387→  const del = useDeleteCustomField();
   388→  const [name, setName] = useState("");
   389→  const [key, setKey] = useState("");
   390→  const [type, setType] = useState("text");
   391→  const [options, setOptions] = useState("");
   392→
   393→  const submit = async () => {
   394→    if (!name.trim() || !key.trim()) return;
   395→    try {
   396→      const opts = type === "select" || type === "multiselect"
   397→        ? options.split(",").map((s) => s.trim()).filter(Boolean)
   398→        : undefined;
   399→      await create.mutateAsync({ name, key, type, options: opts });
   400→      toast.success("Custom field created");
   401→      setName(""); setKey(""); setType("text"); setOptions("");
   402→    } catch (e) { toast.error((e as Error).message); }
   403→  };
   404→  const remove = async (id: string) => {
   405→    try { await del.mutateAsync(id); toast.success("Field deleted"); } catch (e) { toast.error((e as Error).message); }
   406→  };
   407→
