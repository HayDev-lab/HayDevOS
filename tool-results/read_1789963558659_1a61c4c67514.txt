   767→   767→const DICTS: Record<Locale, Dict> = { hy, ru, en };
   768→   768→
   769→   769→/** Translate a key with optional {param} interpolation. */
   770→   770→export function t<T extends string = string>(
   771→   771→  key: string,
   772→   772→  locale: Locale = DEFAULT_LOCALE,
   773→   773→  params?: Record<string, T | number | string>,
   774→   774→): string {
   775→   775→  const dict = DICTS[locale] ?? DICTS[DEFAULT_LOCALE];
   776→   776→  let value = dict[key] ?? DICTS.en[key] ?? key;
   777→   777→  if (params) {
   778→   778→    for (const [k, v] of Object.entries(params)) {
   779→   779→      value = value.replace(new RegExp(`\\{${k}\\}`, "g"), String(v));
   780→   780→    }
   781→   781→  }
   782→   782→  return value;
   783→   783→}
   784→   784→
   785→   785→export function isLocale(value: unknown): value is Locale {
   786→   786→  return value === "hy" || value === "ru" || value === "en";
   787→   787→}

   789→   789→// ─────────────────────────────────────────────────────────────────────────────
   790→   790→// useLocale — Zustand-backed hook (re-exported here so consumers import i18n
   791→   791→// from a single module).
   792→   792→//
   793→   793→// NOTE on circular imports: app-store imports DEFAULT_LOCALE from this module,
   794→   794→// and this module imports useAppStore from app-store. To avoid the cycle
   795→   795→// biting at module-eval time, app-store hardcodes the default locale ("hy")
   796→   796→// instead of importing DEFAULT_LOCALE. That keeps i18n → app-store as the
   797→   797→// only direction, which is safe because useAppStore is only invoked inside
   798→   798→// the hook body (at render time, after all modules are evaluated).
   799→   799→// ─────────────────────────────────────────────────────────────────────────────
   800→   800→
   801→   801→import { useAppStore } from "@/lib/store/app-store";
   802→   802→
   803→   803→export function useLocale() {
   804→   804→  const locale = useAppStore((s: { locale: Locale }) => s.locale);
   805→   805→  const setLocale = useAppStore((s: { setLocale: (l: Locale) => void }) => s.setLocale);
   806→   806→  return {
   807→   807→    locale,
   808→   808→    setLocale,
   809→   809→    t: (key: string, params?: Record<string, string | number>) =>
   810→   810→      t(key, locale, params),
   811→   811→  };
   812→   812→}
