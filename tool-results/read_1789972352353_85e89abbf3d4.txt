    51→const MS_PER_DAY = 86_400_000;
    52→const MS_PER_HOUR = 3_600_000;
    53→
    54→export function resolveWindow(w: TimeWindow, customDays = 14): WindowRange {
    55→  const endMs = Date.now();
    56→  let days: number;
    57→  switch (w) {
    58→    case "today":
    59→      days = 1;
    60→      break;
    61→    case "7d":
    62→      days = 7;
    63→      break;
    64→    case "30d":
    65→      days = 30;
    66→      break;
    67→    case "quarter":
    68→      days = 90;
    69→      break;
    70→    case "custom":
    71→      days = customDays;
    72→      break;
    73→  }
    74→  // For "today" we use a 24h window; for others, day-aligned.
    75→  const startMs = w === "today" ? endMs - MS_PER_HOUR * 24 : endMs - days * MS_PER_DAY;
    76→  const priorStartMs = startMs - (endMs - startMs);
    77→  const labelKey =
    78→    w === "today" ? "control.window.today" :
    79→    w === "7d" ? "control.window.7d" :
    80→    w === "30d" ? "control.window.30d" :
    81→    w === "quarter" ? "control.window.quarter" :
    82→    "control.window.custom";
    83→  return { startMs, endMs, days, priorStartMs, labelKey };
    84→}
    85→
    86→/** True if `iso` falls inside [startMs, endMs). */
    87→function inWindow(iso: string | null | undefined, range: WindowRange): boolean {
    88→  if (!iso) return false;
    89→  const t = new Date(iso).getTime();
    90→  return t >= range.startMs && t < range.endMs;
    91→}
    92→
    93→/** True if `iso` falls inside the prior period [priorStartMs, startMs). */
    94→function inPriorWindow(iso: string | null | undefined, range: WindowRange): boolean {
    95→  if (!iso) return false;
    96→  const t = new Date(iso).getTime();
    97→  return t >= range.priorStartMs && t < range.startMs;
    98→}

   100→/** 7 evenly-spaced buckets across the window. Returns ISO bucket labels + start ms. */
   101→function bucketize(range: WindowRange): { label: string; startMs: number; endMs: number }[] {
   102→  const n = 7;
   103→  const span = range.endMs - range.startMs;
   104→  const step = span / n;
   105→  const buckets: { label: string; startMs: number; endMs: number }[] = [];
   106→  for (let i = 0; i < n; i++) {
   107→    const startMs = range.startMs + i * step;
   108→    const endMs = startMs + step;
   109→    const d = new Date(startMs);
   110→    const label = range.days <= 2
   111→      ? `${d.getHours().toString().padStart(2, "0")}:00`
   112→      : `${d.getMonth() + 1}/${d.getDate()}`;
   113→    buckets.push({ label, startMs, endMs });
   114→  }
   115→  return buckets;
   116→}

   118→/** Deterministic pseudo-history anchored on a real current value. */
   119→function anchoredSparkline(current: number, variance = 0.08): number[] {
   120→  const n = 7;
   121→  // Stable seed from the integer part of current.
   122→  let seed = Math.max(1, Math.floor(Math.abs(current)) + 7);
   123→  const rnd = () => {
   124→    seed = (seed * 9301 + 49297) % 233280;
   125→    return seed / 233280;
   126→  };
   127→  const out: number[] = [];
   128→  for (let i = 0; i < n; i++) {
   129→    // Slight upward drift toward current.
   130→    const drift = (i / (n - 1)) * 0.4 + 0.6;
   131→    const noise = 1 - variance + rnd() * variance * 2;
   132→    out.push(Math.max(0, current * drift * noise));
   133→  }
   134→  out[out.length - 1] = current;
   135→  return out;
   136→}
   137→
   138→/** Count-based sparkline — counts records whose `getDate` falls in each bucket. */
   139→function countSparkline<T>(records: T[], getDate: (r: T) => string | null | undefined, range: WindowRange): number[] {
   140→  const buckets = bucketize(range);
   141→  return buckets.map((b) => records.filter((r) => {
   142→    const t = new Date(getDate(r) ?? "").getTime();
   143→    return t >= b.startMs && t < b.endMs;
   144→  }).length);
   145→}
   146→
   147→/** Delta % between current + prior counts. */
   148→function deltaPct(current: number, prior: number): number {
   149→  if (prior === 0) return current > 0 ? 100 : 0;
   150→  return ((current - prior) / prior) * 100;
   151→}

   384→function buildModuleHealth(): ModuleHealthEntry[] {
   385→  const out: ModuleHealthEntry[] = [];
   386→
   387→  // LeadOS
   388→  const leadsTotal = mockLeads.length;
   389→  const slaBreaches = mockLeads.filter((l) => l.slaBreached === true).length;
   390→  const leadsWarn = mockLeads.filter((l) => l.stage === "new" && !l.firstResponseAt).length;
   391→  out.push({
   392→    moduleId: "leados",
   393→    nameKey: "module.leados",
   394→    health: slaBreaches > 0 ? "critical" : leadsWarn > 0 ? "warning" : "healthy",
   395→    score: 1 - (slaBreaches + leadsWarn * 0.5) / Math.max(1, leadsTotal),
   396→    summary: slaBreaches > 0 ? `${slaBreaches} SLA breach${slaBreaches === 1 ? "" : "es"}` : leadsWarn > 0 ? `${leadsWarn} new leads awaiting response` : "Pipeline healthy",
   397→    counts: { ok: leadsTotal - slaBreaches - leadsWarn, warn: leadsWarn, crit: slaBreaches },
   398→  });
   399→
   400→  // QuoteFlow
   401→  const qTotal = mockQuotes.length;
   402→  const qExpired = mockQuotes.filter((q) => q.status === "expired").length;
   403→  const qExpiring = mockQuotes.filter((q) => {
   404→    if (q.status !== "sent") return false;
   405→    const days = (new Date(q.validUntil).getTime() - Date.now()) / MS_PER_DAY;
   406→    return days <= 7;
   407→  }).length;
   408→  out.push({
   409→    moduleId: "quoteflow",
   410→    nameKey: "module.quoteflow",
   411→    health: qExpired > 0 ? "warning" : qExpiring > 0 ? "warning" : "healthy",
   412→    score: 1 - (qExpired + qExpiring * 0.5) / Math.max(1, qTotal),
   413→    summary: qExpired > 0 ? `${qExpired} expired quote${qExpired === 1 ? "" : "s"}` : qExpiring > 0 ? `${qExpiring} expiring soon` : "Quotes on track",
   414→    counts: { ok: qTotal - qExpired - qExpiring, warn: qExpiring, crit: 0 },
   415→  });
   416→
   417→  // DocSmart
   418→  const dTotal = mockDocuments.length;
   419→  const dBacklog = mockDocuments.filter(
   420→    (d) => d.status === "pending" || d.status === "processing" || d.status === "extracted" || d.status === "classified",
   421→  ).length;
   422→  out.push({
   423→    moduleId: "docsmart",
   424→    nameKey: "module.docsmart",
   425→    health: dBacklog > 4 ? "warning" : dBacklog > 0 ? "warning" : "healthy",
   426→    score: 1 - dBacklog / Math.max(1, dTotal),
   427→    summary: dBacklog > 0 ? `${dBacklog} awaiting review` : "All documents processed",
   428→    counts: { ok: dTotal - dBacklog, warn: dBacklog, crit: 0 },
   429→  });
   430→
   431→  // Autopilot
   432→  const aTotal = mockAutomations.length;
   433→  const aFailing = mockAutomations.filter((a) => a.runs.failed > 0).length;
   434→  const aPaused = mockAutomations.filter((a) => a.status === "paused" || a.status === "draft").length;
   435→  out.push({
   436→    moduleId: "autopilot",
   437→    nameKey: "module.autopilot",
   438→    health: aFailing > 2 ? "critical" : aFailing > 0 ? "warning" : "healthy",
   439→    score: 1 - (aFailing * 0.7 + aPaused * 0.2) / Math.max(1, aTotal),
   440→    summary: aFailing > 0 ? `${aFailing} automation${aFailing === 1 ? "" : "s"} with failures` : `${aTotal - aPaused} active automations`,
   441→    counts: { ok: aTotal - aFailing - aPaused, warn: aPaused, crit: aFailing },
   442→  });
   443→
   444→  // ERP Hub
   445→  const invTotal = mockInvoices.length;
   446→  const invOverdue = mockInvoices.filter((i) => i.status === "overdue").length;
   447→  const lowStock = mockProducts.filter((p) => p.stock <= 12).length;
   448→  out.push({
   449→    moduleId: "erphub",
   450→    nameKey: "module.erphub",
   451→    health: invOverdue > 0 ? "critical" : lowStock > 0 ? "warning" : "healthy",
   452→    score: 1 - (invOverdue + lowStock * 0.4) / Math.max(1, invTotal),
   453→    summary: invOverdue > 0 ? `${invOverdue} overdue invoice${invOverdue === 1 ? "" : "s"}` : lowStock > 0 ? `${lowStock} low-stock SKU${lowStock === 1 ? "" : "s"}` : "Finances healthy",
   454→    counts: { ok: invTotal - invOverdue, warn: lowStock, crit: invOverdue },
   455→  });
   456→
   457→  // Connect
   458→  const iTotal = mockIntegrations.length;
   459→  const iConnected = mockIntegrations.filter((i) => i.status === "connected").length;
   460→  const iDegraded = mockIntegrations.filter((i) => i.status === "degraded" || i.status === "reauth_required").length;
   461→  const iError = mockIntegrations.filter((i) => i.status === "error" || i.status === "disconnected").length;
   462→  out.push({
   463→    moduleId: "connect",
   464→    nameKey: "module.connect",
   465→    health: iError > 0 ? "critical" : iDegraded > 0 ? "warning" : "healthy",
   466→    score: iTotal > 0 ? iConnected / iTotal : 1,
   467→    summary: iError > 0 ? `${iError} integration${iError === 1 ? "" : "s"} down` : iDegraded > 0 ? `${iDegraded} need re-auth` : `${iConnected}/${iTotal} connected`,
   468→    counts: { ok: iConnected, warn: iDegraded, crit: iError },
   469→  });
   470→
   471→  return out;
   472→}

  1444→// AI insights (window-aware — older insights filtered for narrow windows)
  1445→// ─────────────────────────────────────────────────────────────────────────────
  1446→
  1447→export function getAiInsights(window: TimeWindow): { window: WindowRange; insights: AiInsight[] } {
  1448→  const range = resolveWindow(window);
  1449→  const all = buildAiInsights();
  1450→  // For narrow windows (today/7d), drop insights older than the window.
  1451→  const insights = range.days <= 7
  1452→    ? all.filter((i) => new Date(i.ts).getTime() >= range.startMs)
  1453→    : all;
  1454→  return { window: range, insights: insights.length > 0 ? insights : all };
  1455→}
