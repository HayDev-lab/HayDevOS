    96→function RecordPaymentDialog({
    97→  open,
    98→  onOpenChange,
    99→  invoice,
   100→}: {
   101→  open: boolean;
   102→  onOpenChange: (open: boolean) => void;
   103→  invoice: ErpInvoice | null;
   104→}) {
   105→  const { t } = useLocale();
   106→  const [amount, setAmount] = useState<number>(0);
   107→  const [method, setMethod] = useState<PaymentMethod>("bank");
   108→  const [reference, setReference] = useState<string>("");
   109→  const [confirmOpen, setConfirmOpen] = useState(false);
   110→
   111→  const outstanding = invoice ? invoiceOutstanding(invoice) : 0;
   112→  const effectiveAmount = amount || outstanding;
   113→
   114→  if (!invoice) return null;
   115→
   116→  const handleApply = () => {
   117→    if (effectiveAmount <= 0) {
   118→      toast.error(t("erp.payment.amountEnter"));
   119→      return;
   120→    }
   121→    setConfirmOpen(true);
   122→  };
   123→
   124→  const handleConfirm = () => {
   125→    toast.success(t("erp.toast.paymentRecorded", { number: invoice.number }));
   126→    setAmount(0);
   127→    setReference("");
   128→    setMethod("bank");
   129→    onOpenChange(false);
   130→  };
   131→
   132→  return (
   133→    <>
   134→      <Dialog open={open} onOpenChange={onOpenChange}>
   135→        <DialogContent className="sm:max-w-md">
   136→          <DialogHeader>
   137→            <DialogTitle className="flex items-center gap-2">
   138→              <CreditCard className="h-4 w-4 text-lime" />
   139→              {t("erp.invoice.recordPayment")}
   140→            </DialogTitle>
   141→            <DialogDescription>
   142→              {invoice.number} · {customerName(invoice.customerId)}
   143→            </DialogDescription>
   144→          </DialogHeader>
   145→
   146→          <div className="space-y-3">
   147→            <div className="rounded-md border border-border bg-muted/40 p-3 text-sm">
   148→              <div className="flex items-center justify-between text-muted-foreground">
   149→                <span>{t("erp.invoice.amount")}</span>
   150→                <span className="font-medium text-foreground">{formatCurrency(invoice.amount, invoice.currency)}</span>
   151→              </div>
   152→              <div className="mt-1 flex items-center justify-between text-muted-foreground">
   153→                <span>{t("erp.invoice.paid")}</span>
   154→                <span className="font-medium text-success">{formatCurrency(invoice.paidAmount, invoice.currency)}</span>
   155→              </div>
   156→              <Separator className="my-2" />
   157→              <div className="flex items-center justify-between">
   158→                <span className="text-foreground">{t("erp.invoice.outstanding")}</span>
   159→                <span className="font-semibold text-amber">{formatCurrency(outstanding, invoice.currency)}</span>
   160→              </div>
   161→            </div>
   162→
   163→            <div className="grid grid-cols-2 gap-3">
   164→              <div className="space-y-1.5">
   165→                <Label>{t("erp.payment.amount")}</Label>
   166→                <Input
   167→                  type="number"
   168→                  min={0}
   169→                  step="0.01"
   170→                  value={amount || ""}
   171→                  placeholder={outstanding.toFixed(2)}
   172→                  onChange={(e) => setAmount(Math.max(0, Number(e.target.value) || 0))}
   173→                />
   174→              </div>
   175→              <div className="space-y-1.5">
   176→                <Label>{t("erp.payment.method")}</Label>
   177→                <Select value={method} onValueChange={(v) => setMethod(v as PaymentMethod)}>
   178→                  <SelectTrigger className="w-full"><SelectValue /></SelectTrigger>
   179→                  <SelectContent>
   180→                    <SelectItem value="card">{t("erp.payment.method.card")}</SelectItem>
   181→                    <SelectItem value="bank">{t("erp.payment.method.bank")}</SelectItem>
   182→                    <SelectItem value="cash">{t("erp.payment.method.cash")}</SelectItem>
   183→                    <SelectItem value="crypto">{t("erp.payment.method.crypto")}</SelectItem>
   184→                    <SelectItem value="wallet">{t("erp.payment.method.wallet")}</SelectItem>
   185→                  </SelectContent>
   186→                </Select>
   187→              </div>
   188→            </div>
   189→
   190→            <div className="space-y-1.5">
   191→              <Label>{t("erp.payment.reference")}</Label>
   192→              <Input
   193→                value={reference}
   194→                onChange={(e) => setReference(e.target.value)}
   195→                placeholder="WIRE-XXX / CARD-XXX"
   196→              />
   197→            </div>
   198→          </div>
   199→
   200→          <DialogFooter>
   201→            <Button variant="outline" onClick={() => onOpenChange(false)}>{t("common.cancel")}</Button>
   202→            <Button onClick={handleApply}>
   203→              <CreditCard className="h-4 w-4" />
   204→              {t("erp.invoice.recordPayment")}
   205→            </Button>
   206→          </DialogFooter>
   207→        </DialogContent>
   208→      </Dialog>
   209→
   210→      <HighRiskConfirm
   211→        open={confirmOpen}
   212→        onOpenChange={setConfirmOpen}
   213→        actionLabel={`${t("erp.invoice.recordPayment")}: ${invoice.number} · ${formatCurrency(effectiveAmount, invoice.currency)} (${method})`}
   214→        details={
   215→          <div className="space-y-1">
   216→            <div>Invoice: <span className="font-mono">{invoice.number}</span></div>
   217→            <div>Customer: <span className="font-mono">{customerName(invoice.customerId)}</span></div>
   218→            <div>Amount: <span className="font-mono">{formatCurrency(effectiveAmount, invoice.currency)}</span></div>
   219→            <div>Method: <span className="font-mono">{method}</span></div>
   220→            <div>Reference: <span className="font-mono">{reference || "—"}</span></div>
   221→          </div>
   222→        }
   223→        tone="primary"
   224→        onConfirm={handleConfirm}
   225→      />
   226→    </>
   227→  );
   228→}

   479→        onConfirm={() => {
   480→          toast.success(t("erp.toast.invoiceRefunded", { number: invoice.number }));
   481→        }}
   482→      />
   483→    </>

   490→
   491→export function InvoicesView() {
   492→  const { t, locale } = useLocale();
   493→  const [search, setSearch] = useState("");
   494→  const [statusFilter, setStatusFilter] = useState<string>("all");
   495→  const [selected, setSelected] = useState<ErpInvoice | null>(null);
   496→  const [open, setOpen] = useState(false);
   497→
   498→  const filtered = useMemo(() => {
   499→    return erpInvoices.filter((inv) => {
   500→      if (statusFilter !== "all" && inv.status !== statusFilter) return false;
   501→      if (search.trim()) {
   502→        const q = search.trim().toLowerCase();
   503→        const hay = `${inv.number} ${customerName(inv.customerId)}`.toLowerCase();
   504→        if (!hay.includes(q)) return false;
   505→      }
   506→      return true;
   507→    }).sort((a, b) => +new Date(b.createdAt) - +new Date(a.createdAt));
   508→  }, [search, statusFilter]);
   509→
   510→  return (
   511→    <div className="space-y-4">
   512→      <ErpSectionHeader
   513→        icon={Receipt}
   514→        title={t("erp.invoice.title")}
   515→        subtitle={`${filtered.length} / ${erpInvoices.length}`}
   516→        right={<AiProtectedBadge withTooltip />}
   517→      />
   518→
   519→      <Card className="surface-elevated">
   520→        <CardContent className="gap-0 p-0">
   521→          <div className="flex flex-wrap items-center gap-2 border-b border-border px-4 py-3">
   522→            <div className="relative flex-1 min-w-[200px]">
   523→              <Search className="pointer-events-none absolute left-3 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   524→              <Input
   525→                value={search}
   526→                onChange={(e) => setSearch(e.target.value)}
   527→                placeholder={t("erp.invoice.search")}
   528→                className="pl-9"
   529→              />
   530→            </div>
   531→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   532→              <SelectTrigger className="h-9 w-[160px]">
   533→                <SelectValue placeholder={t("erp.invoice.filterStatus")} />
   534→              </SelectTrigger>
   535→              <SelectContent>
   536→                <SelectItem value="all">{t("common.all")}</SelectItem>
   537→                <SelectItem value="draft">{t("erp.invoice.status.draft")}</SelectItem>
   538→                <SelectItem value="sent">{t("erp.invoice.status.sent")}</SelectItem>
   539→                <SelectItem value="paid">{t("erp.invoice.status.paid")}</SelectItem>
   540→                <SelectItem value="overdue">{t("erp.invoice.status.overdue")}</SelectItem>
   541→                <SelectItem value="cancelled">{t("erp.invoice.status.cancelled")}</SelectItem>
   542→              </SelectContent>
   543→            </Select>
   544→          </div>
   545→
   546→          <div className="max-h-[640px] overflow-auto">
   547→            <Table>
   548→              <TableHeader className="sticky top-0 z-10 bg-card">
   549→                <TableRow className="hover:bg-transparent">
   550→                  <TableHead>{t("erp.invoice.number")}</TableHead>
   551→                  <TableHead>{t("erp.invoice.customer")}</TableHead>
   552→                  <TableHead className="hidden md:table-cell">{t("erp.invoice.issueDate")}</TableHead>
   553→                  <TableHead className="hidden lg:table-cell">{t("erp.invoice.dueDate")}</TableHead>
   554→                  <TableHead className="text-right">{t("erp.invoice.amount")}</TableHead>
   555→                  <TableHead className="text-right">{t("erp.invoice.paid")}</TableHead>
   556→                  <TableHead className="text-right hidden md:table-cell">{t("erp.invoice.balance")}</TableHead>
   557→                  <TableHead>{t("erp.invoice.status")}</TableHead>
   558→                  <TableHead className="w-10"></TableHead>
   559→                </TableRow>
   560→              </TableHeader>
   561→              <TableBody>
   562→                {filtered.length === 0 ? (
   563→                  <TableRow>
   564→                    <TableCell colSpan={9}><ErpEmptyState /></TableCell>
   565→                  </TableRow>
   566→                ) : (
   567→                  filtered.map((inv) => {

   569→                    const aging = invoiceAgingBucket(inv);
   570→                    const agingTone =
   571→                      aging === "current" ? "success" :
   572→                      aging === "1_30" ? "warning" :
   573→                      aging === "31_60" ? "warning" :
   574→                      "destructive";
   575→                    const agingCls = toneClasses(agingTone);
   576→                    const isOverdue = inv.status === "overdue";
   577→                    return (
   578→                      <TableRow
   579→                        key={inv.id}
   580→                        onClick={() => { setSelected(inv); setOpen(true); }}
   581→                        className={cn("cursor-pointer", isOverdue && "bg-rose/5 hover:bg-rose/10")}
   582→                      >
   583→                        <TableCell className="font-medium">{inv.number}</TableCell>
   584→                        <TableCell className="text-sm text-foreground">{customerName(inv.customerId)}</TableCell>
   585→                        <TableCell className="hidden text-sm text-muted-foreground md:table-cell">
   586→                          {formatDate(inv.issueDate, locale)}

   588→                        <TableCell className="hidden text-sm text-muted-foreground lg:table-cell">
   589→                          {formatDate(inv.dueAt, locale)}
   590→                        </TableCell>
   591→                        <TableCell className="text-right text-sm font-medium">{formatCurrency(inv.amount, inv.currency)}</TableCell>

   593→                        <TableCell className={cn("hidden text-right text-sm font-medium md:table-cell", agingCls.text)}>
   594→                          {formatCurrency(outstanding, inv.currency)}
   595→                        </TableCell>
   596→                        <TableCell>
   597→                          <StatusBadge status={inv.status} label={t(`erp.invoice.status.${inv.status}` as const)} />
   598→                        </TableCell>
   599→                        <TableCell>
   600→                          <ArrowRight className="h-3.5 w-3.5 text-muted-foreground" />
   601→                        </TableCell>
   602→                      </TableRow>
   603→                    );
   604→                  })
   605→                )}
   606→              </TableBody>
   607→            </Table>
   608→          </div>
   609→        </CardContent>
   610→      </Card>
   611→
   612→      <InvoiceDetail invoice={selected} open={open} onOpenChange={setOpen} />
   613→    </div>
   614→  );
   615→}
